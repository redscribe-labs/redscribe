# including docxtpl "subdocuments" for rich multi-block content (code
# blocks, images, tables) styled using the uploaded template's own named

# Python OOXML zip/XML manipulation, no system dependency (unlike

docxcompose==2.2.0

# Scan import (apps.findings.importers.burp/nmap) parses fully untrusted
# XML from whatever scanner produced it — stdlib xml.etree.ElementTree has
# no DTD/entity-expansion hardening on its own, so a small crafted DOCTYPE
# can cheaply blow up in-memory size well past what the upload itself
# cost. defusedxml.ElementTree is a drop-in replacement (same parse API)
# that forbids DTDs/external entities/entity expansion outright, raising
# instead of processing them. Tiny, single-purpose, the de facto standard
# fix for this exact class of issue.
defusedxml==0.7.1
